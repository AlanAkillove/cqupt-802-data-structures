**1、** 有一段代码：

```c
int brute(int a[], int N)
{
    int i,j,k,m;
    for(i=0;i<N;i++)
        for(j=i+1;j<N;j++)
            for(k=j+1;k<N;k++)
                for(m=k+1;m<N;m++)
                    if(a[i]+a[j]+a[k]+a[m]==0) 
                        return 1;
    return 0;
}
```

```c
struct node{
    int value;
    struct node *next;
};

void rearrange(struct node *list) 
{
    struct node *p, *q;
    int temp;
    if((!list)||!list->next)
        return;
    p=list;
    q=list->next;
    while(q) 
    {
        temp=p->value;
        p->value=q->value;
        q->value=temp;
        p=q->next;
        q=p?p->next:0;
    }
}
```
